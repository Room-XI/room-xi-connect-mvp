// server/services/parentInvite.ts  (NEW)
// ======================================================================
import crypto from "crypto";
import { db } from "../db.js";
import { parentInvites, parents, parentLinks } from "../schema.consent.extras.js";
import { eq } from "drizzle-orm";
import { sendGuardianVerificationEmail } from "./email.provider.js";

export async function ensureParentInviteForYouth(userId: string, parentEmail: string, youthName?: string) {
  // WHY: Ensure only one live invite; avoids spam/duplicates.
  const token = crypto.randomBytes(32).toString("hex");
  const expiresAt = new Date(Date.now() + 30*24*60*60*1000);

  // Upsert invite
  await db.insert(parentInvites).values({ userId, email: parentEmail.toLowerCase(), token, expiresAt }).onConflictDoUpdate({
    target: parentInvites.email,
    set: { userId, token, expiresAt }
  });

  const link = `${process.env.PUBLIC_BASE_URL}/parent/accept/${token}`;
  await sendGuardianVerificationEmail({ guardianEmail: parentEmail, youthName: youthName || "Your youth", verifyUrl: link });
}

export async function acceptParentInvite(token: string, relation = "guardian") {
  const [invite] = await db.select().from(parentInvites).where(eq(parentInvites.token, token));
  if (!invite || invite.expiresAt < new Date()) throw new Error("Invite invalid/expired");

  // Idempotent: create/find parent
  const email = invite.email.toLowerCase();
  const [parent] = await db.insert(parents).values({ email }).onConflictDoNothing().returning();
  const parentId = parent?.id ?? (await db.select().from(parents).where(eq(parents.email, email)))[0].id;

  // Link to youth
  await db.insert(parentLinks).values({ parentId, userId: invite.userId, relation, verifiedAt: new Date() }).onConflictDoNothing();

  // Mark accepted
  await db.update(parentInvites).set({ acceptedAt: new Date() }).where(eq(parentInvites.id, invite.id));
  return { parentId, userId: invite.userId };
}

// ======================================================================
import express from "express";
import { db } from "../db.js";
import { consentRequests, consentsTable } from "../schema.consent.js";
import { parentLinks } from "../schema.consent.extras.js";
import { eq, and } from "drizzle-orm";
import { requireAuth } from "../middleware/security.ts";
import { DateTime } from "luxon";
import { sendGuardianVerificationEmail } from "../services/email.provider.js";

const router = express.Router();

// WHY: Create platform-level consent requests at first sign-in.
router.post("/platform-bootstrap", requireAuth, async (req, res) => {
  const userId = req.session.userId!;
  const scopes = ['platform_use','roomxi_data'];

  const expiresAt = DateTime.now().plus({ days: 30 }).toJSDate();
  for (const scope of scopes) {
    await db.insert(consentRequests).values({
      programId: req.body.programId || req.body.defaultProgramId || req.body.programId ?? "00000000-0000-0000-0000-000000000000", // optional placeholder
      userId,
      scope: [scope],
      status: "pending",
      expiresAt,
    });
  }

  // Notify any linked/soon-to-be-linked parents (email already sent via invite)
  res.json({ ok: true });
});

// WHY: Trigger program consent at RSVP/first check-in.
router.post("/program-request", requireAuth, async (req, res) => {
  const userId = req.session.userId!;
  const { programId, scope = ['registration'] } = req.body ?? {};
  if (!programId) return res.status(400).json({ error: "programId required" });

  // check if consent exists (not revoked)
  const existing = await db.select().from(consentsTable)
    .where(and(eq(consentsTable.userId, userId), eq(consentsTable.programId, programId)));
  if (existing.length && !existing[0].revokedAt) return res.json({ alreadyConsented: true });

  const expiresAt = DateTime.now().plus({ days: 30 }).toJSDate();
  await db.insert(consentRequests).values({ programId, userId, scope, status: "pending", expiresAt });

  // Notify parents (email optional if dashboard exists)
  const linked = await db.select().from(parentLinks).where(eq(parentLinks.userId, userId));
  // send notifications to each linked parent email (left as an exercise; dashboard will show it)
  res.json({ ok: true });
});

export default router;

// server/routes/demographics.js  (NEW) — split youth vs parent-perception
// ======================================================================
import express from "express";
import { z } from "zod";
import { db } from "../db.js";
import { youthDemographics, parentDemographics, parentLinks } from "../schema.consent.extras.js";
import { eq, and } from "drizzle-orm";
import { requireAuth } from "../middleware/security.ts";

const router = express.Router();

const Schema = z.record(z.any());

// Youth writes ONLY their own demographics; parents cannot read this.
router.post("/youth", requireAuth, async (req, res) => {
  const userId = req.session.userId!;
  const parse = Schema.safeParse(req.body);
  if (!parse.success) return res.status(422).json({ error: parse.error.flatten() });

  const [row] = await db.insert(youthDemographics).values({
    userId, answers: parse.data
  }).onConflictDoUpdate({
    target: youthDemographics.userId,
    set: { answers: parse.data, updatedAt: new Date() }
  }).returning();

  res.json({ saved: true, updatedAt: row.updatedAt });
});

// Parent writes their own perception per youth; youth cannot read this.
router.post("/parent", async (req, res) => {
  const parentId = req.session.parentId;
  if (!parentId) return res.status(401).json({ error: "Parent auth required" });

  const { userId, answers } = req.body ?? {};
  if (!userId) return res.status(400).json({ error: "userId required" });

  // verify link
  const link = await db.select().from(parentLinks).where(and(eq(parentLinks.parentId, parentId), eq(parentLinks.userId, userId)));
  if (!link.length) return res.status(403).json({ error: "Not linked to youth" });

  const parse = Schema.safeParse(answers);
  if (!parse.success) return res.status(422).json({ error: parse.error.flatten() });

  await db.insert(parentDemographics).values({ parentId, userId, answers: parse.data }).onConflictDoUpdate({
    target: [parentDemographics.parentId, parentDemographics.userId],
    set: { answers: parse.data, updatedAt: new Date() }
  });

  res.json({ saved: true });
});

export default router;
