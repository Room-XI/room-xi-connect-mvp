// Inputs: last 7 days, each day is a set of check-ins
const moods = ['cold','stormy','foggy','clear','breezy','aurora'] as const;

function weeklyRatios(days: Day[]): Record<Mood, number> {
  const pd: Record<Mood, number>[] = days.map(d => {
    const total = d.checkins.length || 1;
    const counts = countByMood(d.checkins);
    const out: any = {};
    for (const m of moods) out[m] = counts[m] / total;
    return out;
  });
  const sum: any = {}; for (const m of moods) sum[m] = 0;
  for (const day of pd) for (const m of moods) sum[m] += day[m];
  const ratios: any = {}; for (const m of moods) ratios[m] = sum[m] / 7;
  return ratios;
}

function sectorAngles(ratio: Record<Mood, number>, eps = 0.05) {
  const base = Object.fromEntries(moods.map(m => [m, ratio[m] + eps]));
  const total = Object.values(base).reduce((a,b)=>a+b,0);
  const norm = Object.fromEntries(Object.entries(base).map(([m,v])=>[m, v/total]));
  let start = 0;
  const sectors = moods.map(m => {
    const span = 360 * norm[m];
    const s = { mood: m, start, end: start + span };
    start += span;
    return s;
  });
  return sectors;
}
