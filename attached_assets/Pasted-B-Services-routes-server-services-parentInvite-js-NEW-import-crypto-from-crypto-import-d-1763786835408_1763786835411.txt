server/services/parentInvite.js (NEW)

import crypto from "crypto";
import { db } from "../db.js";
import { parentInvites, parents, parentLinks } from "../schema.consent.extras.js";
import { eq } from "drizzle-orm";
import { sendGuardianVerificationEmail } from "../services/email.provider.js";

export async function createOrRefreshParentInvite(userId, parentEmail, youthName) {
  const token = crypto.randomBytes(32).toString("hex");
  const expiresAt = new Date(Date.now() + 30*24*60*60*1000);

  await db.insert(parentInvites).values({ userId, email: parentEmail.toLowerCase(), token, expiresAt })
    .onConflictDoUpdate({ target: parentInvites.email, set: { userId, token, expiresAt } });

  const link = `${process.env.PUBLIC_BASE_URL}/parent/accept/${token}`;
  await sendGuardianVerificationEmail({ guardianEmail: parentEmail, youthName: youthName || "Your youth", verifyUrl: link });
}

export async function acceptParentInvite(token, relation = "guardian") {
  const [invite] = await db.select().from(parentInvites).where(eq(parentInvites.token, token));
  if (!invite || invite.expiresAt < new Date()) throw new Error("Invite invalid/expired");

  const email = invite.email.toLowerCase();
  const [created] = await db.insert(parents).values({ email }).onConflictDoNothing().returning();
  const parentId = created?.id ?? (await db.select().from(parents).where(eq(parents.email, email)))[0].id;

  await db.insert(parentLinks).values({ parentId, userId: invite.userId, relation, verifiedAt: new Date() }).onConflictDoNothing();
  await db.update(parentInvites).set({ acceptedAt: new Date() }).where(eq(parentInvites.id, invite.id));
  return { parentId, userId: invite.userId };
}


server/routes/parent-auth.js (NEW)

import express from "express";
import { acceptParentInvite } from "../services/parentInvite.js";

const router = express.Router();

router.get("/accept/:token", async (req, res) => {
  try {
    const { parentId } = await acceptParentInvite(req.params.token);
    req.session.parentId = parentId;
    res.redirect("/parent");
  } catch {
    res.status(400).send("Invalid or expired invite.");
  }
});

export default router;

server/routes/demographics.js (NEW)

import express from "express";
import { z } from "zod";
import { db } from "../db.js";
import { youthDemographics, parentDemographics, parentLinks } from "../schema.consent.extras.js";
import { and, eq } from "drizzle-orm";

const router = express.Router();
const Schema = z.record(z.any());

const requireYouth = (req, res, next) => req.session?.user ? next() : res.status(401).json({ error: "Not authenticated" });
const requireParent = (req, res, next) => req.session?.parentId ? next() : res.status(401).json({ error: "Parent auth required" });

// Youth private
router.post("/youth", requireYouth, async (req, res) => {
  const userId = req.session.user.id;
  const parsed = Schema.safeParse(req.body);
  if (!parsed.success) return res.status(422).json({ error: parsed.error.flatten() });

  await db.insert(youthDemographics).values({ userId, answers: parsed.data })
    .onConflictDoUpdate({ target: youthDemographics.userId, set: { answers: parsed.data, updatedAt: new Date() } });
  res.json({ saved: true });
});

// Parent perception
router.post("/parent", requireParent, async (req, res) => {
  const parentId = req.session.parentId;
  const { userId, answers } = req.body || {};
  if (!userId) return res.status(400).json({ error: "userId required" });

  const link = await db.select().from(parentLinks).where(and(eq(parentLinks.parentId, parentId), eq(parentLinks.userId, userId)));
  if (!link.length) return res.status(403).json({ error: "Not linked to youth" });

  const parsed = Schema.safeParse(answers);
  if (!parsed.success) return res.status(422).json({ error: parsed.error.flatten() });

  await db.insert(parentDemographics).values({ parentId, userId, answers: parsed.data })
    .onConflictDoUpdate({ target: [parentDemographics.parentId, parentDemographics.userId],
      set: { answers: parsed.data, updatedAt: new Date() } });

  res.json({ saved: true });
});

export default router;


server/routes/mood-tasks.js (NEW)

import express from "express";
import { db } from "../db.js";
import { moodTasks } from "../schema.moodtasks.js";
import { and, eq, isNull } from "drizzle-orm";

const router = express.Router();
const requireAuth = (req, res, next) => req.session?.user ? next() : res.status(401).json({ error: "Not authenticated" });

router.get("/", requireAuth, async (req, res) => {
  const userId = req.session.user.id;
  const rows = await db.select().from(moodTasks).where(and(eq(moodTasks.userId, userId), isNull(moodTasks.completedAt)));
  res.json({ tasks: rows });
});

router.post("/:id/complete", requireAuth, async (req, res) => {
  await db.update(moodTasks).set({ completedAt: new Date() }).where(eq(moodTasks.id, req.params.id));
  res.json({ ok: true });
});

export default router;
