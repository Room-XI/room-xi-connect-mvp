# path: server/src/middleware/auth.ts
import type { Request, Response, NextFunction } from 'express';

export function requireAuth(req: Request, res: Response, next: NextFunction) {
  if (!req.session?.user) return res.status(401).json({ error: 'Unauthorized' });
  next();
}

export function requireRole(...roles: string[]) {
  return (req: Request, res: Response, next: NextFunction) => {
    const r = (req.session as any)?.user?.role;
    if (!r || !roles.includes(r)) return res.status(403).json({ error: 'Forbidden' });
    next();
  };
}

import { Router } from 'express';
import { DateTime } from 'luxon';
import { z } from 'zod';
import { q } from '../db.js';
import { requireAuth } from '../middleware/auth.js';

export const checkinsRouter = Router();
const body = z.object({ mood: z.number().int().min(1).max(6) });

checkinsRouter.use(requireAuth);

checkinsRouter.post('/', async (req, res) => {
  const parse = body.safeParse(req.body);
  if (!parse.success) return res.status(400).json({ error: 'Invalid mood' });
  const uid = (req.session as any).user.id;
  const day = DateTime.now().setZone('America/Edmonton').toISODate();
  try {
    await q(
      `insert into checkins (user_id, date, mood) values ($1,$2,$3)
       on conflict (user_id, date) do update set mood=excluded.mood`,
      [uid, day, parse.data.mood]
    );
    res.json({ ok: true, date: day, mood: parse.data.mood });
  } catch (e) {
    res.status(500).json({ error: 'DB error' });
  }
});

checkinsRouter.get('/streak', async (req, res) => {
  const uid = (req.session as any).user.id;
  const rows = (await q<{ date: string }>(
    `select date from checkins where user_id=$1 order by date desc limit 60`, [uid]
  )).rows;
  const today = DateTime.now().setZone('America/Edmonton').startOf('day');
  let streak = 0;
  for (let i = 0; i < rows.length; i++) {
    const d = DateTime.fromISO(rows[i].date, { zone: 'America/Edmonton' });
    if (d.equals(today.minus({ days: streak }))) streak++; else break;
  }
  res.json({ streak });
});

import { Router } from 'express';

  youth_id: z.string(),
  parent_email: z.string().email(),

});

  const p = inviteSchema.safeParse(req.body);
  if (!p.success) return res.status(400).json({ error: 'Invalid' });
  const token = uuid();
  await q(
    `insert into consent_records (youth_id, parent_email, token, status, province)
     values ($1,$2,$3,'pending',$4)`, [p.data.youth_id, p.data.parent_email, token, p.data.province]

  const token = String(req.body?.token ?? '');
  const rec = (await q<any>(`select * from consent_records where token=$1`, [token])).rows[0];
  if (!rec || rec.status !== 'pending') return res.status(400).json({ error: 'Invalid token' });
  await q(`update consent_records set status='granted', granted_at=now() where token=$1`, [token]);
  res.json({ ok: true });

import nodemailer from 'nodemailer';
import { guardianInviteHtml } from '../../emails/guardianInvite.html.js';
import { guardianInviteText } from '../../emails/guardianInvite.text.js';

export async function sendGuardianInvite(email: string, { token, youthId, province }:
  { token: string; youthId: string; province: string; }) {
  const transporter = nodemailer.createTransport({
    host: process.env.MAIL_HOST, port: Number(process.env.MAIL_PORT ?? 587),
    secure: process.env.MAIL_SECURE === 'true',
    auth: { user: process.env.MAIL_USER, pass: process.env.MAIL_PASS }
  });
  const link = `${process.env.CLIENT_ORIGIN}/consent?token=${encodeURIComponent(token)}`;
  const infoLink = `${process.env.CLIENT_ORIGIN}/about`;
  const html = guardianInviteHtml({ link, infoLink, province, youthId });
  const text = guardianInviteText({ link, infoLink, province, youthId });
  await transporter.sendMail({
    from: process.env.MAIL_FROM,
    to: email,
    subject: 'Room XI — Consent invitation & information',
    text, html
  });
}

async function run() {
  const dir = path.resolve('server/migrations');
  const files = readdirSync(dir).filter(f => f.endsWith('.sql')).sort();
  for (const f of files) {
    const sql = readFileSync(path.join(dir, f), 'utf8');

  user_id text primary key,
  data jsonb not null default '{}'::jsonb,

  province text not null,

  created_at timestamp default now()

create table if not exists disclosure_requests (
  id bigserial primary key,
  parent_id text not null,
  youth_id text not null,
  status text not null check (status in ('pending','approved','denied')),
  decided_at timestamp,
  created_at timestamp default now()
);

create table if not exists checkins (
  user_id text not null,
  date date not null,
  mood int not null check (mood between 1 and 6),
  primary key (user_id, date)
);

create table if not exists programs (
  id uuid default gen_random_uuid() primary key,
  name text not null,
  tags text[] not null default '{}',
  starts_at timestamp not null default now(),

);

create table if not exists program_events (
  id uuid default gen_random_uuid() primary key,

let state: AuthState = 'loading';
export async function isSignedIn(): Promise<boolean> {
  if (!sb) { state = 'authed'; return true; }
  const { data } = await sb.auth.getSession();
  const ok = Boolean(data?.session);
  state = ok ? 'authed' : 'guest';
  if (!ok) logger.warn('Not signed in');
  return ok;
}
export function getAuthState(): AuthState { return state; }
export async function getAuthHeader(): Promise<Record<string,string>> {
  if (!sb) return {};
  const { data } = await sb.auth.getSession();
  const tok = data?.session?.access_token;
  return tok ? { Authorization: `Bearer ${tok}` } : {};
}

# path: src/lib/api.ts
import { API_BASE } from './featureFlags';
import { getAuthHeader, isSignedIn } from './auth';
import { logger } from './logger';

export class ApiError extends Error { status: number; body?: unknown;
  constructor(msg: string, status: number, body?: unknown){ super(msg); this.status=status; this.body=body; } }

async function withAuth(init?: RequestInit): Promise<RequestInit> {
  const hdr = await getAuthHeader();
  return { ...init, headers: { 'X-Requested-With':'fetch', ...(init?.headers||{}), ...hdr }, credentials: init?.credentials ?? 'include' };
}

export async function apiFetch<T>(path: string, init?: RequestInit, retryOnCsrf = true): Promise<T> {
  if (!(await isSignedIn())) throw new ApiError('Not signed in', 401);
  const res = await fetch(`${API_BASE}${path}`, await withAuth(init));
  if (res.status === 419 && retryOnCsrf) {
    const retry = await fetch(`${API_BASE}${path}`, await withAuth(init));
    if (retry.ok) return retry.json() as Promise<T>;
  }
  if (!res.ok) {
    let body: unknown; try { body = await res.json(); } catch {}
    logger.error('API error', path, res.status, body);
    throw new ApiError(`Request failed: ${path}`, res.status, body);
  }
  return res.json() as Promise<T>;
}

export function useAuthGuard() {
  const [ready, setReady] = useState(false);
  const [authed, setAuthed] = useState(false);
  useEffect(() => {
    let mounted = true;
    (async () => {
      const ok = await isSignedIn();
      if (!mounted) return;
      setAuthed(ok);
      setReady(true);
    })();
    return () => { mounted = false; };
  }, []);

# Project Guide (Canonical)
- **Stack**: Vite (web), Express + Postgres (API), Sessions + CSRF, Nodemailer (email).
- **MVP**: Mood check-in & orb, Crisis support, Program finder, Universal intake with youth/parent split.
- **Auth & Visibility**: Youth demographics private by default; guardian disclosure is request + youth approval.
- **Canada**: CASL-compliant emails, Quebec <14 parental consent, data residency preference in Canada.
- **Security**: Helmet/CSP, CORS allowlist, CSRF, rate limits, role checks, strict TS & ESLint, no `eval`/`innerHTML`.
- **Retention**: `server/src/jobs/retention.ts` enforces schedule; document per-table durations.
- **Run**: `npm run migrate` → `npm run dev`. Build with `npm run build`. Start with `npm start`.
- **QA**: a11y (axe, keyboard), e2e (auth, check-in, programs, consent, disclosure).
